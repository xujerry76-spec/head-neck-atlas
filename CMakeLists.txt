cmake_minimum_required(VERSION 3.16.3...3.19.7 FATAL_ERROR)
project(HeadNeckAtlas)
find_package(Slicer REQUIRED)
include(${Slicer_USE_FILE})
set(EXTENSION_HOMEPAGE "https://github.com/Slicer/Slicer")
set(EXTENSION_CATEGORY "Research")
set(EXTENSION_CONTRIBUTORS "HeadNeckAtlas contributors")
set(EXTENSION_DESCRIPTION "Local head and neck CT research workflow with TotalSegmentator, editable segmentation and slice labels.")
set(EXTENSION_DEPENDS "TotalSegmentator")
add_subdirectory(HeadNeckAtlas)
include(${Slicer_EXTENSION_GENERATE_CONFIG})
include(${Slicer_EXTENSION_CPACK})
